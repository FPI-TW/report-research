# research schema 的 CHECK 約束清單（golden）。由 tests/test_schema_constraints.py 對帳。
#
# 緣起：導入 Alembic 前 db/schema.sql 只靠 `CREATE TABLE IF NOT EXISTS`，改了 CHECK 在既有庫
# （＝生產）永遠不會生效、`make schema` 卻照樣成功。現在約束變更寫在 revision 裡，既有庫與空庫
# 跑同一段 SQL；這份 golden 仍逐行釘住「空庫 `alembic upgrade head` 之後該有的約束」，讓約束
# 變更在 diff 裡被看見。
#
# 格式：`表名.約束名 = pg_get_constraintdef(oid)`，以 Python `sorted()` 排序（不用 SQL
# 的 ORDER BY，避免 collation locale 影響順序）。定義是 PostgreSQL 反解析出來的字面，
# 不是 schema.sql 的原文——`x IN (...)` 會被寫成 `x = ANY (ARRAY[...])`、括號也會多一層。
#
# 要更新（只在你**刻意**改了約束時）：
#   REPORT_MARK_WRITE_CONSTRAINTS=1 uv run pytest -q tests/test_schema_constraints.py
# （REPORT_MARK_DB_URL 指向剛 upgrade head 的全新空庫），把 diff 與 revision 一起提交。
app_user.app_user_role_check = CHECK ((role = ANY (ARRAY['admin'::text, 'user'::text])))
extraction_log.extraction_log_stopped_at_check = CHECK ((stopped_at = ANY (ARRAY['ingested'::text, 'skip_admin'::text, 'scanned'::text, 'not_research'::text, 'extract_error'::text])))
report_signal.report_signal_extraction_status_check = CHECK ((extraction_status = ANY (ARRAY['pending'::text, 'valid'::text, 'partial'::text, 'rejected'::text])))
report_signal.report_signal_rating_normalized_check = CHECK ((rating_normalized = ANY (ARRAY['buy'::text, 'overweight'::text, 'neutral'::text, 'underweight'::text, 'sell'::text, 'unknown'::text])))
report_takeaway.ck_report_takeaway_span = CHECK (((quote_start IS NULL) OR ((quote_start >= 0) AND (quote_end > quote_start))))
report_takeaway.report_takeaway_anchor_method_check = CHECK (((anchor_method IS NULL) OR (anchor_method = ANY (ARRAY['exact'::text, 'normalized'::text, 'prefix'::text]))))
report_takeaway.report_takeaway_extraction_status_check = CHECK ((extraction_status = ANY (ARRAY['pending'::text, 'valid'::text, 'partial'::text, 'rejected'::text])))
user_scope.user_scope_scope_check = CHECK ((scope = ANY (ARRAY['qa_content.read'::text, 'ops.operate'::text])))
