# research schema 的 CHECK 約束清單（golden）。由 tests/test_schema_constraints.py 對帳。
#
# 這份檔案關掉的破口：db/schema.sql 用的是 `CREATE TABLE IF NOT EXISTS`，對**既有**表
# 是完全的 no-op。新增欄位有 `ALTER TABLE ... ADD COLUMN IF NOT EXISTS` 補，但 CHECK
# 約束沒有對應的補丁——所以改了 schema.sql 裡的 CHECK，在既有庫（＝生產）上永遠不會
# 生效，而 `make schema` 照樣安靜地成功。這裡逐行釘住「一個乾淨的庫套完 schema.sql
# 之後該有的約束」，任何變更都會在 CI 變紅，逼你同時寫出給既有庫用的 ALTER。
#
# 格式：`表名.約束名 = pg_get_constraintdef(oid)`，以 Python `sorted()` 排序（不用 SQL
# 的 ORDER BY，避免 collation locale 影響順序）。定義是 PostgreSQL 反解析出來的字面，
# 不是 schema.sql 的原文——`x IN (...)` 會被寫成 `x = ANY (ARRAY[...])`、括號也會多一層。
#
# 要更新（只在你**刻意**改了約束時）：
#   REPORT_MARK_WRITE_CONSTRAINTS=1 uv run pytest -q tests/test_schema_constraints.py
# 然後把 diff 一起提交，並確認生產庫也套了對應的 ALTER。
report_run.report_run_status_check = CHECK ((status = ANY (ARRAY['queued'::text, 'retrieving'::text, 'outlining'::text, 'drafting'::text, 'verifying'::text, 'rendering'::text, 'completed'::text, 'failed'::text, 'cancelled'::text])))
report_section.report_section_status_check = CHECK ((status = ANY (ARRAY['pending'::text, 'retrieving'::text, 'drafting'::text, 'drafted'::text, 'verifying'::text, 'final'::text, 'failed'::text])))
report_signal.report_signal_extraction_status_check = CHECK ((extraction_status = ANY (ARRAY['pending'::text, 'valid'::text, 'partial'::text, 'rejected'::text])))
report_signal.report_signal_rating_normalized_check = CHECK ((rating_normalized = ANY (ARRAY['buy'::text, 'overweight'::text, 'neutral'::text, 'underweight'::text, 'sell'::text, 'unknown'::text])))
